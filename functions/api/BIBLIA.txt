export async function onRequest(context) {
// 1. Pega a chave que você salvou em Variáveis e Segredos
const apiKey = context.env.GEMINI_API_KEY;

if (!apiKey) {
return new Response(JSON.stringify({ error: "Chave GEMINI_API_KEY não encontrada nas configurações do Cloudflare." }), {
status: 500,
headers: { "Content-Type": "application/json" }
});
}

// Permite apenas requisições POST do seu site
if (context.request.method !== "POST") {
return new Response(JSON.stringify({ error: "Método não permitido" }), {
status: 405,
headers: { "Content-Type": "application/json" }
});
}

try {
// 2. Pega a mensagem/prompt que o seu frontend mandou
const body = await context.request.json();
const prompt = body.prompt || body.mensagem;

if (!prompt) {
return new Response(JSON.stringify({ error: "Nenhum texto foi enviado para o Gemini." }), {
status: 400,
headers: { "Content-Type": "application/json" }
});
}

// 3. Faz a requisição direto para a API oficial do Gemini (usando o modelo Flash, que é rápido)
const geminiUrl = https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey};

const apiResponse = await fetch(geminiUrl, {
method: "POST",
headers: { "Content-Type": "application/json" },
body: JSON.stringify({
contents: [{ parts: [{ text: prompt }] }]
})
});

const data = await apiResponse.json();

// 4. Devolve a resposta do Gemini para a sua página
return new Response(JSON.stringify(data), {
status: 200,
headers: { "Content-Type": "application/json" }
});

} catch (err) {
return new Response(JSON.stringify({ error: err.message }), {
status: 500,
headers: { "Content-Type": "application/json" }
});
}
}
